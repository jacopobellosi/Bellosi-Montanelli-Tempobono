title [UC2] – Log In

participant User
participant "BBP (App)" as BBP
participant "BBP Server" as SRV

User->BBP: openLogin()
BBP-->User: showLoginForm()

User->BBP: submitCredentials(identifier, password)
activate BBP
BBP->(2)SRV: authenticate(identifier, password)
activate SRV
SRV->SRV: lookupUser(identifier)
SRV->SRV: verifyPassword(password, storedHash)

alternative Invalid credentials
  SRV-->(2)BBP: AuthResult(FAILED, INVALID_CREDENTIALS)
  BBP-->User: showError("Invalid username or password")
  note over User,BBP: User remains on the login page
or Valid credentials
  SRV->SRV: createSession(userId, deviceInfo)
  SRV->SRV: issueToken(userId, sessionId)
  SRV-->(2)BBP: AuthResult(SUCCESS, token)
  BBP-->User: redirectToMainMap()
  note over User,BBP: Role becomes Registered User
end
deactivate SRV
deactivate BBP
