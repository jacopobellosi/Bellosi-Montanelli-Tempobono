title Design Sequence Diagram: UC1 Register Account

actor User
participant "UI layer"
participant "Network layer"
participant "Local storage"
participant "Controller layer"
participant "user service"
participant "Hash password"
participant "Token manager"
participant "Model"
database "BBP Database"

# --- Inizio Flusso ---
User->"UI layer": Enters details (email, pass) & Taps "Register"
activate User
activate "UI layer"

"UI layer"->"Network layer": register(email, password, name)
activate "Network layer"

# 1. Chiamata API
"Network layer"->"Controller layer": POST /api/auth/register
activate "Controller layer"

# 2. Validazione e Routing
"Controller layer"->"Controller layer": validateJsonFormat()
"Controller layer"->"user service": registerUser(userData)
activate "user service"

# 3. Verifica Unicità (Auth Service chiama Model)
"user service"->"Model": findUserByEmail(email)
activate "Model"
"Model"->"BBP Database": SELECT id FROM users WHERE email = ?
activate "BBP Database"
"BBP Database"-->"Model": result (empty if new)
deactivate "BBP Database"
"Model"-->"user service": userNotFound (null)
deactivate "Model"

# --- Blocco Alternativo (Alt) ---
alt User already exists
    "user service"-->"Controller layer": Error: UserAlreadyExistsException
    "Controller layer"-->"Network layer": 409 Conflict
    "Network layer"-->"UI layer": Show Error "Email already taken"
else New User (Success Flow)
    
    # 4. Hashing Password
    "user service"->"Hash password": hash(password)
    activate "Hash password"
    "Hash password"-->"user service": passwordHash
    deactivate "Hash password"

    # 5. Salvataggio
    "user service"->"Model": createUser(userData, passwordHash)
    activate "Model"
    "Model"->"BBP Database": INSERT INTO users (...) VALUES (...)
    activate "BBP Database"
    "BBP Database"-->"Model": success (userId)
    deactivate "BBP Database"
    "Model"-->"user service": userEntity (userId)
    deactivate "Model"

    # 6. Generazione Token
    "user service"->"Token manager": generateToken(userId, role)
    activate "Token manager"
    "Token manager"-->"user service": jwtToken
    deactivate "Token manager"

    # 7. Risposta al Controller
    "user service"-->"Controller layer": UserCreated(userDto, token)
    deactivate "user service"

    # 8. Risposta al Client
    "Controller layer"-->"Network layer": 201 Created (JSON with JWT)
    deactivate "Controller layer"

    # 9. Salvataggio Locale
    "Network layer"->"Local storage": saveToken(jwtToken)
    activate "Local storage"
    "Local storage"-->"Network layer": ok
    deactivate "Local storage"

    # 10. Redirect UI
    "Network layer"-->"UI layer": success
    deactivate "Network layer"
    "UI layer"-->User: Redirect to Map / Main Screen
    deactivate "UI layer"
    deactivate User
end
