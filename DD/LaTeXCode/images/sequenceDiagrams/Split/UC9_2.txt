@startuml
title Design Sequence Diagram: UC9 Save Trip (Part 2: Orchestration)

actor User
participant "UI layer"
participant "Network layer"
participant "Local storage"
participant "Controller layer"
participant "Token manager"
participant "Path adding"
participant "Add obstacle to path"
participant "Add status to path"
participant "Trip adding"

' --- DEFINIZIONE REPOSITORY SPECIFICI (CON ALIAS) ---
participant "Path Repository\n(Model)" as PathRepo
participant "Report Repository\n(Model)" as ReportRepo
participant "Trip Repository\n(Model)" as TripRepo

database "BBP Database"

activate User
activate "UI layer"

note right of User: Flow continues from Part 1 (User Tapped Save)

# Recupero Dati Locali
"UI layer"->"Local storage": getSavedObstacles()
activate "Local storage"
"Local storage"-->"UI layer": confirmedObstacles
deactivate "Local storage"

# Invio Dati al Backend
"UI layer"->"Network layer": savePathAndTripSession(fullData, token)
activate "Network layer"
"Network layer"->"Controller layer": POST /api/sessions
activate "Controller layer"

# A. Sicurezza
"Controller layer"->"Token manager": validateToken(token)
activate "Token manager"
"Token manager"-->"Controller layer": tokenValid (userId)
deactivate "Token manager"

# B. Orchestrazione: Salvataggio PATH + REPORT (Step 1)
"Controller layer"->"Path adding": createPath(geometry, name, visibility)
activate "Path adding"

    # B1. Salva Geometria Path (Usa PathRepo)
    "Path adding"->PathRepo: savePath(geometry, name, visibility)
    activate PathRepo
    PathRepo->"BBP Database": INSERT INTO paths...
    activate "BBP Database"
    "BBP Database"-->PathRepo: newPathId
    deactivate "BBP Database"
    PathRepo-->"Path adding": pathId
    deactivate PathRepo
    "Path adding"-->"Controller layer": pathId
    deactivate "Path adding"

    # B2. Salva Ostacoli (Usa ReportRepo)
    loop For each confirmed obstacle
        "Controller layer"->"Add obstacle to path": saveObstacle(pathId, type, location)
        activate "Add obstacle to path"
        "Add obstacle to path"->ReportRepo: createReport(obstacleData)
        activate ReportRepo
        ReportRepo->"BBP Database": INSERT INTO reports...
        activate "BBP Database"
        "BBP Database"-->ReportRepo: Confirm
        deactivate "BBP Database"
        ReportRepo-->"Add obstacle to path": Success
        deactivate ReportRepo
        "Add obstacle to path"-->"Controller layer": ok
        deactivate "Add obstacle to path"
    end

    # B3. Salva Status (Usa ReportRepo)
    "Controller layer"->"Add status to path": saveStatus(pathId, statusValue)
    activate "Add status to path"
    "Add status to path"->ReportRepo: createReport(statusData)
    activate ReportRepo
    ReportRepo->"BBP Database": INSERT INTO reports...
    activate "BBP Database"
    "BBP Database"-->ReportRepo: Confirm
    deactivate "BBP Database"
    ReportRepo-->"Add status to path": Success
    deactivate ReportRepo
    "Add status to path"-->"Controller layer": ok
    deactivate "Add status to path"

# C. Orchestrazione: Salvataggio TRIP (Step 2)
"Controller layer"->"Trip adding": createTrip(stats, weather, pathId, userId, visibility)
activate "Trip adding"
# Usa TripRepo
"Trip adding"->TripRepo: saveTrip(stats, weather, pathId, visibility)
activate TripRepo
TripRepo->"BBP Database": INSERT INTO trips...
activate "BBP Database"
"BBP Database"-->TripRepo: newTripId
deactivate "BBP Database"
TripRepo-->"Trip adding": tripId
deactivate TripRepo
"Trip adding"-->"Controller layer": tripId
deactivate "Trip adding"

# D. Risposta Finale
"Controller layer"-->"Network layer": 201 Created
deactivate "Controller layer"

"Network layer"->"Local storage": clearTempTrip()
activate "Local storage"
"Local storage"-->"Network layer": success
deactivate "Local storage"
"Network layer"-->"UI layer": Success
deactivate "Network layer"
"UI layer"-->User: Redirect to My Trips
deactivate "UI layer"
deactivate User
@enduml
