# --- Inizio Flusso ---
User->"UI layer": Taps on a specific Path
activate User
activate "UI layer"



"UI layer"->"Network layer": getPathDetails(pathId, token)
activate "Network layer"

# 1. Chiamata API
"Network layer"->"Controller layer": GET /api/paths/{id}
activate "Controller layer"



# 3. Routing
"Controller layer"->"Detailed path view": getPathDetails(pathId)
activate "Detailed path view"

# 4. Recupero Dati Completi dal DB
"Detailed path view"->"Model": getFullPathData(pathId)
activate "Model"
"Model"->"BBP Database": SELECT * FROM paths JOIN gps_points JOIN reports...
activate "BBP Database"
"BBP Database"-->"Model": fullPathData (geometry, reports, etc.)
deactivate "BBP Database"
"Model"-->"Detailed path view": fullPathData
deactivate "Model"

# 5. Calcolo Stato (On-Read)
"Detailed path view"->"Path merger": calculateMetrics(reports)
activate "Path merger"
"Path merger"-->"Detailed path view": currentStatus
deactivate "Path merger"

# 6. Recupero Meteo (Forecast) - R14
"Detailed path view"->"Weather API client": getForecast(lat, long)

activate "Weather API client"
"Weather API client"->"External weather service": getWeather(lat, long)
activate "External weather service"
"External weather service"-->"Weather API client":weatherData
deactivate "External weather service"
"Weather API client"-->"Detailed path view": weatherForecast
deactivate "Weather API client"

# 7. Risposta
"Detailed path view"-->"Controller layer": pathDetailsDTO
deactivate "Detailed path view"

"Controller layer"-->"Network layer": 200 OK (JSON)
deactivate "Controller layer"

"Network layer"-->"UI layer": Display Path Details
deactivate "Network layer"
"UI layer"-->User: Shows Map, Stats, Status, Weather
