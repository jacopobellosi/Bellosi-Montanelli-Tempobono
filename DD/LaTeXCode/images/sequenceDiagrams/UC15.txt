title Design Sequence Diagram: UC15 Submit Obstacle Report

actor User
participant "UI layer"
participant "Network layer"
participant "Local storage"
participant "Controller layer"
participant "Token manager"
participant "Add obstacle to path"
participant "Model"
database "BBP Database"

# --- Precondizione ---
# L'utente è in UC5 (Visualizzazione) o UC11 (Creazione Manuale)

# --- Fase 1: Interazione Utente ---
User->"UI layer": Taps "Report Obstacle"
activate "UI layer"
"UI layer"-->User: Show Form (Type, Location Pin)

User->"UI layer": Selects Type (e.g. Pothole), Places Pin
User->"UI layer": Taps "Submit"

# --- Fase 2: Gestione Contestuale ---
alt Context: Called from UC11 (Draft Mode)
    # Salvataggio Locale (Bozza)
    "UI layer"->"Local storage": saveDraftObstacle(type, location)
    activate "Local storage"
    "Local storage"-->"UI layer": saved
    deactivate "Local storage"
    "UI layer"-->User: Show "Obstacle added to draft"
    note right of "UI layer": Data will be sent to server\nlater during UC12 (Save Path)

else Context: Called from UC5 (Live Mode)
    # Salvataggio Immediato su Server
    "UI layer"->"Local storage": getToken()
    activate "Local storage"
    "Local storage"-->"UI layer": jwtToken
    deactivate "Local storage"

    "UI layer"->"Network layer": submitObstacleReport(pathId, type, location, token)
    activate "Network layer"

    # 1. Chiamata API
    "Network layer"->"Controller layer": POST /api/reports/obstacle
    activate "Controller layer"

    # 2. Sicurezza
    "Controller layer"->"Token manager": validateToken(token)
    activate "Token manager"
    "Token manager"-->"Controller layer": valid (userId)
    deactivate "Token manager"

    # 3. Routing al modulo specifico
    "Controller layer"->"Add obstacle to path": saveObstacle(pathId, userId, type, location)
    activate "Add obstacle to path"

    # 4. Persistenza (Geospaziale)
    "Add obstacle to path"->"Model": createObstacleReport(data)
    activate "Model"
    "Model"->"BBP Database": INSERT INTO reports (type='OBSTACLE', geom=Point(...))
    activate "BBP Database"
    "BBP Database"-->"Model": success
    deactivate "BBP Database"
    "Model"-->"Add obstacle to path": reportId
    deactivate "Model"

    "Add obstacle to path"-->"Controller layer": reportId
    deactivate "Add obstacle to path"

    # 5. Risposta
    "Controller layer"-->"Network layer": 201 Created
    deactivate "Controller layer"

    "Network layer"-->"UI layer": Show "Obstacle Reported"
    deactivate "Network layer"
end
deactivate "UI layer"
