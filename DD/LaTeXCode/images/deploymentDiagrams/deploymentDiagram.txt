' ==========================================

left to right direction

' --- 1. Presentation Tier (Client) ---
package "Presentation Tier" <<Presentation>> {
    node "User Smartphone" as Phone <<PresentationDevice>> {
        node "iOS / Android" as PhoneOS <<PresentationOS>> {
            artifact "Mobile App" <<Artifact>> as ClientApp
            artifact "Maps App" <<Artifact>> as MapsApp
        }
    }
}

' --- 2. Application Tier (Server Infrastructure) ---
package "Application Tier" <<Application>> {
    
    node "Firewall" as FW <<AppDevice>>
    
    node "Web Server" as WebSrv <<AppDevice>> {
        node "Linux (OS)" as WebOS <<AppOS>> {
            node "Nginx" as WebEnv <<AppEnv>> {
                artifact "Load Balancer" <<Artifact>> as LB
            }
        }
    }

    node "Application Server" as AppSrv <<AppDevice>> {
        node "Linux (OS)" as AppOS <<AppOS>> {
            node "JVM / Docker" as AppEnv <<AppEnv>> {
                artifact "BBP Backend Server" <<Artifact>> as Backend
            }
        }
    }
}

' --- 3. Data Tier (Database - FIX DEFINITIVO) ---
package "Data Tier" <<Data>> {
    ' Sintassi semplificata per evitare nodi fantasma
    node "Database Server" as DBSrv <<DataDevice>> {
        node "Linux (OS)" as DBOS <<DataOS>> {
            node "PostgreSQL Engine" as DBEnv <<DataEnv>> {
                artifact "BBP Schema\n(PostGIS)" <<Artifact>> as DB
            }
        }
    }
}

' --- 4. External Services (Remote) ---
node "Weather Server" as WeatherSrv <<ExternalDevice>> {
    node "Server OS" as WeatherOS <<ExternalOS>> {
        artifact "Weather API" <<Artifact>> as Weather
    }
}

' ==========================================
'               CONNESSIONI
' ==========================================

' 1. Client -> Navigazione Locale
ClientApp --> MapsApp : OS Intent

' 2. Client -> Firewall
ClientApp --> FW : HTTPS

' 3. Firewall -> Web Server
FW --> LB : HTTPS

' 4. Web Server -> App Server
LB --> Backend : HTTP

' 5. App Server -> Database
Backend --> DB : JDBC/SQL (Port 5432)

' 6. App Server -> Meteo Esterno
Backend --> Weather : HTTPS (REST)
